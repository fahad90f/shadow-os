# Shadow OS boot menu theme (GRUB gfxmenu): the "Conduit Lock" instrument at rest.
#
# This file is written by hand. The pictures and fonts next to it are made by
# assets/src/boot/gen-grub-theme.sh; docs/architecture/boot-menu.md explains
# the layout and the rules below.
#
# GRUB cannot animate a menu. The only parts that change are the three
# elements with id "__timeout__" (ring of dots, T-NN number, line with GRUB's
# own "Ns remaining." text): they step once a second and all disappear at the
# first key press. Everything else is still, and none of it mentions the
# countdown, because it stays on screen after that key press.
#
# Rules (each one breaks the menu in real GRUB when ignored):
#   - no '%' in a label text except the one "%02d" of the countdown number;
#   - font names must be the exact names inside the .pf2 files;
#   - the terminal box must hold 80x24 cells of the terminal font plus its
#     border, or GRUB replaces it with a centred box that covers the header.
#
# Everything sits in one column 616 px wide, placed with "percent +- pixels",
# so that it also fits a 640x480 screen.

title-text: ""
desktop-image: "background.png"
desktop-image-scale-method: "stretch"
desktop-color: "#04060B"

# Command line (c) and entry editor (e). "GNU Unifont Regular 16" is the font
# GRUB always has, also under UEFI Secure Boot. The box starts under the
# instrument and covers the entry frame and the key hints. From 712 px screen
# height up it holds the 80x24 cells of that font (640x384 plus the border).
# A lower screen has no room for such a box under the instrument: GRUB then
# puts its own centred box over the header (seen at 800x600) or uses the
# whole screen (seen at 640x480).
terminal-font: "GNU Unifont Regular 16"
terminal-box: "term_*.png"
terminal-left: "50%-330"
terminal-top: "4%+194"
terminal-width: "660"
terminal-height: "50%+32"
terminal-border: "2"

# Countdown, part 1: 40 dots on the track (radius 84 px), taken off clockwise

    tick_bitmap = "tick.png"

# The name, as a picture (left of the instrument).

# Countdown, part 2: the remaining seconds (right of the instrument), centred
# over the conduit's right run (144 px). Centred, because under Secure Boot
# the number is drawn with the small built-in font and would hang at one end.
+ label {
    id = "__timeout__"
    left = 50%+100
    top = 4%+75
    width = 144
    align = "center"
    text = "T-%02d"
    font = "Shadow Mono Bold 28"
    color = "#22D3EE"
}

# The entries. Rows are 40 px high with 6 px between them; the row pictures
# are single centre slices of exactly the row's width (616 - 2 * 12 = 592), so
# the selected row sits exactly on its slot. Icons are 32x20 pictures whose
# left 12 px are empty (GRUB draws them at the row's left edge, where the
# selection bar is). GRUB cuts a long title at the row's right edge and has
# no padding there, so the row pictures draw nothing at text height right of
# the icon.
